define void @double(ptr nonnull %0) {
entry:
  %load = load i32, ptr %0, align 4
  %multiply = mul i32 %load, 2
  store i32 %multiply, ptr %0, align 4
  ret void
}

  store i32 5, ptr %value, align 4
  call void @double(ptr %value)

}
