source_filename = "CaracalModule"
target triple = "x86_64-unknown-none"

define i32 @func() {
entry:
  %i = alloca i32, align 4
  store i32 0, ptr %i, align 4
  br label %while_condition

while_condition:                                  ; preds = %if_end, %entry
  br i1 true, label %while_loop, label %while_end

while_loop:                                       ; preds = %while_condition
  %i1 = load i32, ptr %i, align 4
  %i2 = load i32, ptr %i, align 4
  %addtmp = add i32 %i2, 1
  store i32 %addtmp, ptr %i, align 4
  %i3 = load i32, ptr %i, align 4
  %getmp = icmp sge i32 %i3, 3
  br i1 %getmp, label %if_true, label %if_end

if_true:                                          ; preds = %while_loop
  br label %while_end

if_end:                                           ; preds = %while_loop
  br label %while_condition

while_end:                                        ; preds = %if_true, %while_condition
  %i4 = load i32, ptr %i, align 4
  ret i32 %i4
}
