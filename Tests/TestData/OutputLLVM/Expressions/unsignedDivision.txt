define float @unsignedDivide(i8 %0, i8 %1) {
entry:
  %int_to_float = uitofp i8 %0 to float
  %int_to_float1 = uitofp i8 %1 to float
  %divide = fdiv float %int_to_float, %int_to_float1
  ret float %divide
}

define float @signedDivide(i32 %0, i32 %1) {
entry:
  %int_to_float = sitofp i32 %0 to float
  %int_to_float1 = sitofp i32 %1 to float
  %divide = fdiv float %int_to_float, %int_to_float1
  ret float %divide
}

define float @foldedHighByteDivision() {
entry:
  ret float 1.000000e+02
}
