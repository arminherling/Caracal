define void @func() {
entry:
  %x = alloca i32, align 4
  store i32 1, ptr %x, align 4
  %load = load i32, ptr %x, align 4
  %less_or_equal = icmp sle i32 %load, 2
  br i1 %less_or_equal, label %if.true, label %if.continuation

if.true:                                          ; preds = %entry
  ret void

if.continuation:                                  ; preds = %entry
  ret void
}
