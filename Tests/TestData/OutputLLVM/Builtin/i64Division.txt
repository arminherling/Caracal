define i32 @main() {
entry:
  br i1 true, label %and.rhs, label %and.merge

and.rhs:                                          ; preds = %entry
  br label %and.merge

and.merge:                                        ; preds = %and.rhs, %entry
  %phi = phi i1 [ true, %entry ], [ true, %and.rhs ]
  br i1 %phi, label %if.true, label %if.continuation

if.true:                                          ; preds = %and.merge
  ret i32 1

if.continuation:                                  ; preds = %and.merge
  ret i32 0
}
