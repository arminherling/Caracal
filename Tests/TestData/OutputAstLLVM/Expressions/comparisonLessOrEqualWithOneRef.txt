define void @func() {
entry:
  %x = alloca i32, align 4
  %y = alloca i32, align 4
  store i32 1, ptr %x, align 4
  store i32 2, ptr %y, align 4
  %y1 = load i32, ptr %y, align 4
  %load = load i32, ptr %x, align 4
  %letmp = icmp sle i32 %load, %y1
  br i1 %letmp, label %if_true, label %if_end

if_true:                                          ; preds = %entry
  ret void

if_end:                                           ; preds = %entry
  ret void
}
